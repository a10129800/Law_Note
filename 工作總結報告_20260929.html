3. 智慧錨點監聽 (Hash Listener)

當 URL 帶有特定小節錨點（如 `#sec-intro-concept`）自動穿透至內文並平滑滾動；若無錨點則精準停泊於獨立封面。

JS LOGIC: switchView(view) VANILLA JS 
function switchView(view) {
  // 1. 取得視圖容器節點
  const isCover = (view === 'cover');
  
  // 2. 切換中央主內容容器
  viewCover.classList.toggle('hidden', !isCover);
  viewChapter1.classList.toggle('hidden', isCover);
  
  // 3. 連動右側目錄 TOC 隱現與寬幅拓寬
  rightToc.classList.toggle('hidden', isCover);
  mainContainer.className = isCover 
    ? "lg:col-span-9 xl:col-span-10 space-y-8"
    : "lg:col-span-6 xl:col-span-8 space-y-8";

  // 4. 側邊選單高亮與自動回頂
  updateNavHighlight(view);
  window.scrollTo({ top: 0, behavior: 'smooth' });
}
